create table student156( 
    sname varchar2(50) constraint not_null_sname not null, 
    dob date constraint not_null_dob not null, 
    mob_no number(10) constraint not_null_mob not null, 
    roll_no number(10) constraint pk_roll_no primary key, 
    email varchar2(50) constraint not_null_email not null, 
    pincode number(6) constraint not_null_pincode not null, 
    house_no number(3) constraint not_null_house_no not null, 
    city varchar2(50) constraint not_null_city not null, 
    constraint ck_sname_address unique(sname, pincode, house_no, city), 
    constraint ck_email unique(email), 
    constraint ck_mobile unique(mob_no) 
);

create table section156( 
    batch number(4) constraint not_null_batch not null, 
    sec_id varchar2(2) constraint pk_sec_id primary key 
);

create table faculty156( 
    fname varchar2(20) constraint not_null_fname not null, 
    salary number(10,2) constraint not_null_salary not null, 
    emp_id varchar2(20) constraint pk_emp_id primary key, 
    fmob number(10) constraint not_null_fmob not null, 
    constraint ck_fmob unique(fmob)  
);

create table course156( 
    cid varchar2(20) constraint pk_cid primary key, 
    cname varchar2(50) constraint not_null_cname not null, 
    discipline varchar2(50) constraint not_null_discipline not null, 
    offeredby_dep varchar2(10) constraint not_null_offdep not null, 
    constraint fk_offdep foreign key(offeredby_dep) references department156(dep_id), 
    constraint ck_cname_discipline unique(cname, discipline) 
);


create table subject156( 
    papercode varchar2(20) constraint pk_pcode primary key, 
    pname varchar2(20) constraint not_null_pname not null, 
    credit number(2) constraint not_null_cred not null 
);


create table contains156( 
    ccid varchar2(20), 
    cpapcode varchar2(20), 
    semester number(1), 
    constraint pk_contains primary key(ccid, cpapcode, semester), 
    constraint fk_ccid foreign key(ccid) references course156(cid), 
    constraint fk_cpap foreign key(cpapcode) references subject156(papercode) 
);


create table exam156( 
    perfby_roll number(10), 
    epapcode varchar2(20), 
    esession number(3), 
    marks number(3) constraint not_null_marks not null, 
    constraint pk_exam primary key(perfby_roll, epapcode, esession), 
    constraint fk_perfbyroll foreign key(perfby_roll) references student156(roll_no), 
    constraint fk_epapcode foreign key(epapcode) references subject156(papercode) 
);


create table teaches156( 
    tempid varchar2(20), 
    tpapcode varchar2(10), 
    tsession number(3), 
    constraint pk_teaches primary key(tempid, tpapcode, tsession), 
    constraint fk_tid foreign key(tempid) references faculty156(emp_id), 
    constraint fk_tpap foreign key(tpapcode) references subject156(papercode) 
);

create table instructed156( 
    inssecid varchar2(2), 
    insempid varchar2(20),

    constraint pk_ins primary key(inssecid, insempid, isession), 
    constraint fk_isecid foreign key(inssecid) references section156(sec_id), 
    constraint fk_iempid foreign key(insempid) references faculty156(emp_id) 
);
