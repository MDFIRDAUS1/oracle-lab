-- student

create table student156(
    sname varchar2(50) constraint notnull_sname not null,
    dob date constraint notnull_dob not null,
    mobile number(10) constraint notnull_mob2 not null,
    roll number(10) constraint pkey_roll_stud primary key,
    email varchar2(50) constraint notnull_email2 not null,
    pincode number(6) constraint notnull_pin not null,
    houseno number(3) constraint notnull_hno not null,
    city varchar2(50) constraint notnull_city not null,

    constraint ckey_sname_address
    unique(sname,pincode,houseno,city),

    constraint ckey_email_stud
    unique(email),

    constraint ckey_mob_stud
    unique(mobile)
);


-- section

create table section156(
    studying_cid varchar2(20)
    constraint notnull_scid not null,

    sec_id varchar2(2)
    constraint pkey_sec primary key,

    batch number(4)
    constraint notnull_batch not null
);


-- faculty

create table faculty156(
    fname varchar2(20)
    constraint notnull_fname not null,

    salary number(10,2)
    constraint notnull_sal not null,

    empid varchar2(20)
    constraint pkey_empid primary key,

    emob number(10)
    constraint notnull_emob not null,

    constraint ckey_emob unique(emob)
);

-- course

create table course156(
    cid varchar2(20)
    constraint pkey_cid primary key,

    constraint cname_disc_unique
    unique(cname,discipline),

    constraint cons_fkey_offdep
    foreign key(offeredby_dep)
    references department156(depid)
);

    'SUBJECT156',
    'CONTAINS156',

)
